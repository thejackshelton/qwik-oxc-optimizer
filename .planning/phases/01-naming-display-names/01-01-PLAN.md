```rust
fn escape_sym(str: &str) -> String {
    str.chars()
        .flat_map(|x| match x {
            'A'..='Z' | 'a'..='z' | '0'..='9' => Some(x),
            _ => Some('_'),
        })
        .fold((String::new(), None), |(mut acc, prev), x| {
            if x == '_' {
                if prev.is_none() {
                    (acc, None)  // leading underscore: skip
                } else {
                    (acc, Some('_'))  // store pending underscore
                }
            } else {
                if prev == Some('_') {
                    acc.push('_');  // emit stored underscore
                }
                acc.push(x);
                (acc, Some(x))
            }
        })
        .0
}
```
