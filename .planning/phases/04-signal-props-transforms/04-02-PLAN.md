<objective>
Add loop/iteration tracking infrastructure and implement q:p injection + event handler parameter transformation to match SWC behavior.

Purpose: The OXC optimizer has zero loop tracking infrastructure. SWC maintains `loop_depth`, `iteration_var_stack`, and tracks iteration method callbacks to: (1) inject `q:p`/`q:ps` props on JSX elements whose event handlers use iteration variables, and (2) transform event handler lambdas to accept iteration variables as extra parameters. This accounts for ~22 snapshot diffs.

Note: QRL hoisting (converting inline QRL calls to `const SegmentName = qrl(...)` variable declarations above loops) is deferred to Phase 6. The OXC Traverse architecture stores hoisted data as strings, and injecting into function-level blocks (vs module top level) requires a different mechanism than the existing `hoisted_function_stmts` pattern which only works at module level via `exit_program`. The q:p injection and event handler transforms are the higher-impact changes for snapshot parity.

Output: Loop tracking state in QwikTransform, enter/exit hooks for all loop constructs + iteration methods, q:p injection in JSX transform, and event handler parameter transformation.
</objective>

```rust
fn enter_for_statement(&mut self, node: &mut ForStatement<'a>, _ctx: &mut TraverseCtx<'a, ()>) {
    self.loop_depth += 1;
    // Extract iteration variable from init: for (let i = ...) or for (var i = ...)
    let iteration_vars = if let Some(ForStatementInit::VariableDeclaration(ref decl)) = node.init {
        decl.declarations.first()
            .and_then(|d| {
                if let BindingPatternKind::BindingIdentifier(ref ident) = d.id.kind {
                    Some(vec![ident.name.to_string()])
                } else {
                    None
                }
            })
            .unwrap_or_default()
    } else {
        Vec::new()
    };
    self.iteration_var_stack.push(iteration_vars);
}

fn exit_for_statement(&mut self, _node: &mut ForStatement<'a>, _ctx: &mut TraverseCtx<'a, ()>) {
    self.iteration_var_stack.pop();
    self.loop_depth -= 1;
}

fn enter_for_in_statement(&mut self, node: &mut ForInStatement<'a>, _ctx: &mut TraverseCtx<'a, ()>) {
    self.loop_depth += 1;
    let iteration_vars = match &node.left {
        ForStatementLeft::VariableDeclaration(decl) => {
            decl.declarations.first()
                .and_then(|d| {
                    if let BindingPatternKind::BindingIdentifier(ref ident) = d.id.kind {
                        Some(vec![ident.name.to_string()])
                    } else {
                        None
                    }
                })
                .unwrap_or_default()
        }
        ForStatementLeft::AssignmentTargetIdentifier(ident) => {
            vec![ident.name.to_string()]
        }
        _ => Vec::new(),
    };
    self.iteration_var_stack.push(iteration_vars);
}

fn exit_for_in_statement(&mut self, _node: &mut ForInStatement<'a>, _ctx: &mut TraverseCtx<'a, ()>) {
    self.iteration_var_stack.pop();
    self.loop_depth -= 1;
}
```

```rust
// After the attribute processing loop, before building the final _jsxSorted call:
if loop_depth > 0 && !iteration_vars.is_empty() {
    // Collect which iteration variables are used by event handler expressions in this element.
    // An event handler is identified by its attribute name starting with "on" and ending with "$".
    // The QRL replacement expressions are already stored in the var_props by this point
    // (event handler attributes become q-e:eventname entries in var_props).
    //
    // Check which iteration variables the event handler expressions reference.
    // For each event handler var_prop, check if its expression uses any iteration variable.
    let mut used_iter_vars: Vec<String> = Vec::new();
    for iter_var in iteration_vars {
        // Check if any var_prop expression references this iteration variable.
        // This is a simplified check -- look for the variable name in the var_props values.
        for (key, value_expr) in &var_props {
            if key.starts_with("q-") && expr_uses_ident(value_expr, iter_var) {
                if !used_iter_vars.contains(iter_var) {
                    used_iter_vars.push(iter_var.clone());
                }
                break;
            }
        }
    }

    if !used_iter_vars.is_empty() {
        // Add q:p (single variable) or q:ps (multiple variables) to var_props
        if used_iter_vars.len() == 1 {
            // q:p = iterVar (identifier reference)
            let var_name = &used_iter_vars[0];
            let ident_expr = ctx.ast.expression_identifier(SPAN, ctx.ast.atom(var_name));
            var_props.push(("q:p".to_string(), ident_expr));
        } else {
            // q:ps = [var1, var2, ...] (array expression)
            let mut elements = ctx.ast.vec();
            for var_name in &used_iter_vars {
                let ident = ctx.ast.expression_identifier(SPAN, ctx.ast.atom(var_name));
                elements.push(ArrayExpressionElement::from(ident));
            }
            let arr = ctx.ast.expression_array(SPAN, elements, false);
            var_props.push(("q:ps".to_string(), arr));
        }
    }
}
```

**Step 4: Add expr_uses_ident helper function** in jsx_transform.rs:

```rust
/// Recursively check if an expression references an identifier by name.
fn expr_uses_ident(expr: &Expression<'_>, name: &str) -> bool {
    match expr {
        Expression::Identifier(ident) => ident.name.as_str() == name,
        Expression::StaticMemberExpression(mem) => expr_uses_ident(&mem.object, name),
        Expression::ComputedMemberExpression(mem) => {
            expr_uses_ident(&mem.object, name) || expr_uses_ident(&mem.expression, name)
        }
        Expression::BinaryExpression(bin) => {
            expr_uses_ident(&bin.left, name) || expr_uses_ident(&bin.right, name)
        }
        Expression::CallExpression(call) => {
            expr_uses_ident(&call.callee, name)
                || call.arguments.iter().any(|a| arg_uses_ident(a, name))
        }
        Expression::ConditionalExpression(cond) => {
            expr_uses_ident(&cond.test, name)
                || expr_uses_ident(&cond.consequent, name)
                || expr_uses_ident(&cond.alternate, name)
        }
        Expression::TemplateLiteral(tpl) => {
            tpl.expressions.iter().any(|e| expr_uses_ident(e, name))
        }
        Expression::UnaryExpression(u) => expr_uses_ident(&u.argument, name),
        Expression::LogicalExpression(log) => {
            expr_uses_ident(&log.left, name) || expr_uses_ident(&log.right, name)
        }
        Expression::AssignmentExpression(assign) => expr_uses_ident(&assign.right, name),
        Expression::ParenthesizedExpression(paren) => expr_uses_ident(&paren.expression, name),
        _ => false,
    }
}

fn arg_uses_ident(arg: &Argument<'_>, name: &str) -> bool {
    match arg {
        Argument::SpreadElement(s) => expr_uses_ident(&s.argument, name),
        _ => {
            // Argument inherits Expression variants
            if let Some(expr) = arg.as_expression() {
                expr_uses_ident(expr, name)
            } else {
                false
            }
        }
    }
}
```
