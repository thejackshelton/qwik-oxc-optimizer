**Step 2: Update analyze_props_destructuring to handle all ObjectPatProp variants.**

The current code only handles `BindingProperty` (the OXC equivalent of SWC's ObjectPatProp::KeyValue). OXC's ObjectPattern has a `properties` vec of `BindingProperty` and an optional `rest`. Each `BindingProperty` has a `key` (PropertyKey) and `value` (BindingPattern), and may have a nested AssignmentPattern for defaults.

Examine the OXC AST structure for destructuring patterns:

Pattern `{ count }` -> BindingProperty { key: "count", value: BindingIdentifier("count"), computed: false, shorthand: true }

Pattern `{ count = 3 }` -> BindingProperty { key: "count", value: AssignmentPattern { left: BindingIdentifier("count"), right: NumericLiteral(3) }, shorthand: true }

Pattern `{ count: c }` -> BindingProperty { key: "count", value: BindingIdentifier("c"), shorthand: false }

Pattern `{ count: c = 3 }` -> BindingProperty { key: "count", value: AssignmentPattern { left: BindingIdentifier("c"), right: NumericLiteral(3) }, shorthand: false }

Pattern `{ stuff: { hey } }` -> BindingProperty { key: "stuff", value: ObjectPattern { ... } }

Update the property loop:
```rust
for prop in &obj_pat.properties {
    let key_name = extract_property_key_name(&prop.key);

    match &prop.value.kind {
        BindingPatternKind::BindingIdentifier(ident) => {
            // Simple: { count } or { count: c }
            if let Some(key) = key_name {
                let local = ident.name.to_string();
                info.prop_keys.push((key, local));
            }
        }
        BindingPatternKind::AssignmentPattern(assign) => {
            // Default value: { count = 3 } or { count: c = 3 }
            if let Some(key) = key_name {
                if let BindingPatternKind::BindingIdentifier(ref ident) = assign.left.kind {
                    let local = ident.name.to_string();
                    info.prop_keys.push((key.clone(), local));

                    // Check if default value is const (literal, etc.)
                    if is_const_default_value(&assign.right) {
                        // Serialize the default value for ?? operator
                        let default_code = serialize_expression(&assign.right);
                        info.prop_defaults.push((key, default_code));
                    } else {
                        // Non-const default -> skip entire transform
                        info.skip = true;
                        return info;
                    }
                } else {
                    // Complex left-hand pattern -> skip
                    info.skip = true;
                    return info;
                }
            }
        }
        BindingPatternKind::ObjectPattern(_) | BindingPatternKind::ArrayPattern(_) => {
            // Nested destructuring -> skip entire transform (matches SWC behavior)
            info.skip = true;
            return info;
        }
    }
}
```

Add helper functions:
```rust
/// Check if a default value expression is "const" (safe to inline as ?? operand).
/// Matches SWC's is_const_expr for default values: literals, identifiers known to be const.
fn is_const_default_value(expr: &Expression<'_>) -> bool {
    match expr {
        Expression::NumericLiteral(_) => true,
        Expression::StringLiteral(_) => true,
        Expression::BooleanLiteral(_) => true,
        Expression::NullLiteral(_) => true,
        Expression::Identifier(ident) => {
            matches!(ident.name.as_str(), "undefined" | "NaN" | "Infinity")
        }
        Expression::UnaryExpression(unary) => is_const_default_value(&unary.argument),
        _ => false,
    }
}

/// Serialize an expression to a code string for use in ?? operator.
fn serialize_expression(expr: &Expression<'_>) -> String {
    let mut codegen = oxc::codegen::Codegen::new();
    codegen.print_expression(expr);
    codegen.into_source_text()
}
```
