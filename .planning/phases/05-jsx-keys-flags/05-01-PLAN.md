  let jsx_key_prefix = {
      use std::collections::hash_map::DefaultHasher;
      use std::hash::Hasher;
      use base64::Engine;

      let mut hasher = DefaultHasher::new();

  **IMPORTANT**: Check if the `base64` crate is in Cargo.toml dependencies. If not, add it. The SWC optimizer uses base64 encoding. If adding base64 is problematic, use a manual base64url encoding of the 8 bytes (the output only needs 2 chars so a simple lookup table works).

  Alternative without base64 crate (preferred if base64 not already a dependency):
  ```rust
  let jsx_key_prefix = {
      use std::collections::hash_map::DefaultHasher;
      use std::hash::Hasher;

      let mut hasher = DefaultHasher::new();
      if let Some(ref scope) = options.scope {
          hasher.write(scope.as_bytes());
      }
      hasher.write(filename.as_bytes());
      let file_hash = hasher.finish();

      // Base64url encode the first 2 chars of the LE bytes
      // MUST use URL_SAFE alphabet (matches SWC), NOT standard base64 (+/)
      const CHARS: &[u8] = b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-_";
      let bytes = file_hash.to_le_bytes();
      let b0 = bytes[0] as usize;
      let b1 = bytes[1] as usize;
      let c0 = CHARS[b0 >> 2] as char;
      let c1 = CHARS[((b0 & 0x03) << 4) | (b1 >> 4)] as char;
      // Replace - and _ with 0 (matching SWC)
      let c0 = if c0 == '-' || c0 == '_' { '0' } else { c0 };
      let c1 = if c1 == '-' || c1 == '_' { '0' } else { c1 };
      format!("{}{}", c0, c1)
  };
  ```

  **CRITICAL**: Verify the computed key prefix for "test.tsx" (no scope) produces "u6". If it doesn't, the base64 encoding alphabet or byte order is wrong. The result must be exactly "u6".
