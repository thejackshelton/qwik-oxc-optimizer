  <action>
  The current `build_segment_code_with_hoisted` function (lines ~23-206) emits imports in a hardcoded order:
  1. `_captures` import
  2. `qrl` import
  3. Lazy import declarations
  4. Segment Qrl-suffixed imports
  5. Custom JSX / `_jsxSorted` / `_jsxSplit` / `_fnSignal` / `_wrapProp` etc. (hardcoded order)
  6. User-code imports (from `segment.needed_imports`)

  SWC's approach is fundamentally different: it collects ALL identifiers used in the segment body via `IdentCollector`, sorts them alphabetically, then looks up each identifier in the global imports to determine its source and emit its import. This produces a single alphabetical import list.

  **Refactor approach:** Instead of the current hardcoded if-checks, build a list of (local_name, import_kind, source, original_imported_name) tuples, sort by local_name, then emit.

  **Step 1: Define an ImportEntry struct:**

  ```rust
  /// A single import to emit in a segment module, sortable by local name.
  struct SegmentImportEntry {
      /// The local binding name (e.g., "_jsxSorted", "useStore", "dep3")
      local_name: String,
      /// The import source module (e.g., "@qwik.dev/core", "dep3/something")
      source: String,
      /// The kind of import
      kind: ImportKind,
      /// For aliased imports, the original imported name (e.g., "Fragment" for "_Fragment")
      imported_name: Option<String>,
  }
  ```

  **Step 2: Collect all needed imports into a Vec<SegmentImportEntry>:**

  Replace the hardcoded if-chain with a collection phase:

  ```rust
  let mut imports: Vec<SegmentImportEntry> = Vec::new();
  let core = &options.core_module;
  let jsx_runtime = format!("{}/jsx-runtime", core);

  // _captures (special: always first in SWC when needed, but sorted alphabetically it comes before most)
  if segment.captures && !segment.capture_names.is_empty() {
      imports.push(SegmentImportEntry {
          local_name: "_captures".to_string(),
          source: core.clone(),
          kind: ImportKind::Named,
          imported_name: None,
      });
  }

  // qrl
  if segment.needs_qrl_import {
      imports.push(SegmentImportEntry {
          local_name: "qrl".to_string(),
          source: core.clone(),
          kind: ImportKind::Named,
          imported_name: None,
      });
  }

  // Segment Qrl-suffixed imports
  for qrl_name in &segment.segment_qrl_names {
      imports.push(SegmentImportEntry {
          local_name: qrl_name.clone(),
          source: core.clone(),
          kind: ImportKind::Named,
          imported_name: None,
      });
  }

  // Framework imports detected by body_code.contains()
  // Check custom JSX source first
  if let Some(jsx_source) = custom_jsx_source {
      if body_code.contains("_jsx") {
          imports.push(SegmentImportEntry {
              local_name: "_jsx".to_string(),
              source: format!("{}/jsx-runtime", jsx_source),
              kind: ImportKind::Named,
              imported_name: Some("jsx".to_string()),
          });
      }
  } else if body_code.contains("_jsxSorted") {
      imports.push(SegmentImportEntry {
          local_name: "_jsxSorted".to_string(),
          source: core.clone(),
          kind: ImportKind::Named,
          imported_name: None,
      });
  }
  if body_code.contains("_jsxSplit") {
      imports.push(SegmentImportEntry { local_name: "_jsxSplit".to_string(), source: core.clone(), kind: ImportKind::Named, imported_name: None });
  }
  if body_code.contains("_fnSignal") || !hoisted_stmts.is_empty() {
      imports.push(SegmentImportEntry { local_name: "_fnSignal".to_string(), source: core.clone(), kind: ImportKind::Named, imported_name: None });
  }
  if body_code.contains("_wrapProp") {
      imports.push(SegmentImportEntry { local_name: "_wrapProp".to_string(), source: core.clone(), kind: ImportKind::Named, imported_name: None });
  }
  if body_code.contains("_Fragment") {
      imports.push(SegmentImportEntry { local_name: "_Fragment".to_string(), source: jsx_runtime.clone(), kind: ImportKind::Named, imported_name: Some("Fragment".to_string()) });
  }
  if body_code.contains("inlinedQrl") {
      imports.push(SegmentImportEntry { local_name: "inlinedQrl".to_string(), source: core.clone(), kind: ImportKind::Named, imported_name: None });
  }
  if body_code.contains("_noopQrl") {
      imports.push(SegmentImportEntry { local_name: "_noopQrl".to_string(), source: core.clone(), kind: ImportKind::Named, imported_name: None });
  }
  if body_code.contains("_qrlSync") {
      imports.push(SegmentImportEntry { local_name: "_qrlSync".to_string(), source: core.clone(), kind: ImportKind::Named, imported_name: None });
  }
  if body_code.contains("_getVarProps") {
      imports.push(SegmentImportEntry { local_name: "_getVarProps".to_string(), source: core.clone(), kind: ImportKind::Named, imported_name: None });
  }
  if body_code.contains("_getConstProps") {
      imports.push(SegmentImportEntry { local_name: "_getConstProps".to_string(), source: core.clone(), kind: ImportKind::Named, imported_name: None });
  }
  if body_code.contains("_restProps") {
      imports.push(SegmentImportEntry { local_name: "_restProps".to_string(), source: core.clone(), kind: ImportKind::Named, imported_name: None });
  }
  if body_code.contains("_chk") {
      imports.push(SegmentImportEntry { local_name: "_chk".to_string(), source: core.clone(), kind: ImportKind::Named, imported_name: None });
  }
  if body_code.contains("_val") {
      imports.push(SegmentImportEntry { local_name: "_val".to_string(), source: core.clone(), kind: ImportKind::Named, imported_name: None });
  }

  // User-code imports
  for import_info in &segment.needed_imports {
      for (idx, spec_name) in import_info.specifiers.iter().enumerate() {
          let kind = import_info.specifier_kinds.get(idx).unwrap_or(&ImportKind::Named);
          let imported_name = if matches!(kind, ImportKind::Named) {
              import_info.specifier_aliases.get(spec_name).cloned()
          } else {
              None
          };
          imports.push(SegmentImportEntry {
              local_name: spec_name.clone(),
              source: import_info.source.clone(),
              kind: kind.clone(),
              imported_name,
          });
      }
  }
  ```

  **Step 3: Sort alphabetically by local_name:**

  ```rust
  imports.sort_by(|a, b| a.local_name.cmp(&b.local_name));
  ```

  This matches SWC's `local_idents.sort()` which sorts by `(Atom, SyntaxContext)` -- since we don't have SyntaxContext, plain alphabetical by name is equivalent.

  **Step 4: Emit one import per entry (do NOT merge same-source imports in segments):**

  SWC emits one import per identifier in segment modules (it does NOT merge). So emit individually:

  ```rust
  let mut parts: Vec<String> = Vec::new();

  for entry in &imports {
      match entry.kind {
          ImportKind::Default => {
              parts.push(format!("import {} from \"{}\";", entry.local_name, entry.source));
          }
          ImportKind::Namespace => {
              parts.push(format!("import * as {} from \"{}\";", entry.local_name, entry.source));
          }
          ImportKind::Named => {
              if let Some(ref imported) = entry.imported_name {
                  parts.push(format!("import {{ {} as {} }} from \"{}\";", imported, entry.local_name, entry.source));
              } else {
                  parts.push(format!("import {{ {} }} from \"{}\";", entry.local_name, entry.source));
              }
          }
      }
  }
  ```

  **Step 5: Emit lazy imports AFTER regular imports (they are const declarations, not imports):**

  Lazy import declarations (`const i_XXXX = ...`) go after the sorted imports:
  ```rust
  for (hash, import_path) in &segment.child_lazy_imports {
      parts.push(format!("const i_{} = () => import(\"{}\");", hash, import_path));
  }
  ```

  **Step 6: Emit hoisted functions after lazy imports:**
  ```rust
  for (fn_code, str_code) in hoisted_stmts {
      parts.push(fn_code.clone());
      parts.push(str_code.clone());
  }
  ```

  **Step 7: Emit capture restoration + export:**

  Keep the existing capture injection and export logic unchanged.

  **IMPORTANT:** The `_captures` import is special in SWC. In SWC's code_move.rs, `_captures` is emitted FIRST (before the sorted local_idents loop), not sorted. However, since `_captures` sorts before most other identifiers alphabetically (starts with `_c`), it will naturally come first or near-first when sorted. Check if this matters: `_captures` < `_chk` < `_fnSignal` < `_Fragment` < `_getConstProps` etc. Yes, `_captures` naturally sorts first among framework imports. So alphabetical sorting handles this correctly.
