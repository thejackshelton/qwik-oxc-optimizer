```rust
/// Check if a name is an event handler attribute (on[A-Z]*$ pattern).
pub(crate) fn is_event_handler_attr(name: &str) -> bool {
    let base_name = if let Some(pos) = name.find(':') {
        &name[pos + 1..]
    } else {
        name
    };
    if base_name == "event$" {
        return true;
    }
    if base_name.starts_with("on") && base_name.len() > 3 {
        if let Some(ch) = base_name.chars().nth(2) {
            return ch.is_ascii_uppercase() || ch == '-';
        }
    }
    false
}
```

Then in transform.rs at the JSX prop segment creation site, use this function with the attribute name that's already available in the context.

**Add helper function** (in transform.rs, outside the impl block):

```rust
/// Compute a SourceLocation from byte offsets for diagnostic highlights.
/// Uses +1 offset for SWC BytePos parity (SWC uses 1-based byte positions).
fn compute_highlight_from_span(source: &str, lo: u32, hi: u32) -> crate::types::SourceLocation {
    let (start_line, start_col) = byte_offset_to_line_col(source, lo as usize);
    let (end_line, end_col) = byte_offset_to_line_col(source, hi as usize);
    crate::types::SourceLocation {
        lo: lo + 1,
        hi: hi + 1,
        start_line,
        start_col,
        end_line,
        end_col,
    }
}

/// Convert a byte offset to (1-based line, 0-based column).
fn byte_offset_to_line_col(source: &str, offset: usize) -> (u32, u32) {
    let mut line = 1u32;
    let mut col = 0u32;
    for (i, ch) in source.char_indices() {
        if i >= offset {
            break;
        }
        if ch == '\n' {
            line += 1;
            col = 0;
        } else {
            col += 1;
        }
    }
    (line, col)
}
```

If so, verify that the Qrl counterpart exists in `self.collected.module_level_decls` or `self.collected.exported_local_names`.

**Logic:**
```rust
// After segment creation for a non-core $-call:
let qrl_name = words::dollar_to_qrl_name(&callee_name);
if !self.collected.dollar_imports.contains(&callee_name)  // not a core import

    && !self.collected.exported_local_names.contains(qrl_name.as_str())

    self.diagnostics.push(crate::types::Diagnostic {

    });
}
```
